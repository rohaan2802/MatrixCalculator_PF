cmake_minimum_required(VERSION 3.20)
project(MatrixCalculator_PF VERSION 2.0 LANGUAGES CXX)

add_executable(matrix_calculator src/matrix_calculator.cpp)
target_compile_features(matrix_calculator PRIVATE cxx_std_17)
if(WIN32)
    target_link_libraries(matrix_calculator PRIVATE user32)
endif()
if(MSVC)
    target_compile_options(matrix_calculator PRIVATE /W4 /utf-8)
else()
    target_compile_options(matrix_calculator PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(matrix_regression tests/matrix_regression.cpp)
    target_compile_features(matrix_regression PRIVATE cxx_std_17)
    if(WIN32)
        target_link_libraries(matrix_regression PRIVATE user32)
    endif()
    if(MSVC)
        target_compile_options(matrix_regression PRIVATE /utf-8)
    endif()
    add_test(NAME matrix_regression COMMAND matrix_regression)
endif()
